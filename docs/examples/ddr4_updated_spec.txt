// Example 2: Modular interface-based approach for DDR4 memory
// Define DDR4 interfaces split by functional groups

// Define DDR4 Data Interface
interface DDR4Data {
  parameters {
    is_controller: boolean = true;      // Controller or DRAM role
    voltage: voltage = 1.2V;            // JEDEC standard voltage
    width: int = 16;                    // Data width in bits per component
  }
  
  pins {
    // Data signals
    DQ[0:width-1]: bidir(voltage);      // Data pins
    DQS[0:width/8-1]: bidir(voltage);   // Data strobe (1 per byte)
    DQS_n[0:width/8-1]: bidir(voltage); // Differential data strobe
    DM_DBI[0:width/8-1]: is_controller ? out(voltage) : in(voltage); // Data mask or data bus inversion
  }
  
  // Timing constraints for data interface
  constraints {
    setup(DQ to posedge DQS): >30ps;
    hold(DQ to posedge DQS): >30ps;
    slew_rate(DQ): 3-5V/ns;
  }
}

// Define DDR4 Command Interface
interface DDR4Command {
  parameters {
    is_controller: boolean = true;      // Controller or DRAM role
    voltage: voltage = 1.2V;            // JEDEC standard voltage
    ranks: int = 1;                     // Number of ranks
  }
  
  pins {
    // Command signals
    CS_n[0:ranks-1]: is_controller ? out(voltage) : in(voltage);   // Chip select (per rank)
    ACT_n: is_controller ? out(voltage) : in(voltage);             // Activation command
    RAS_n_A16: is_controller ? out(voltage) : in(voltage);         // Row address strobe / A16 
    CAS_n_A15: is_controller ? out(voltage) : in(voltage);         // Column address strobe / A15
    WE_n_A14: is_controller ? out(voltage) : in(voltage);          // Write enable / A14
    CKE[0:ranks-1]: is_controller ? out(voltage) : in(voltage);    // Clock enable (per rank)
    ODT[0:ranks-1]: is_controller ? out(voltage) : in(voltage);    // On-die termination (per rank)
    RESET_n: is_controller ? out(voltage) : in(voltage);           // Reset
  }
}

// Define DDR4 Address Interface
interface DDR4Address {
  parameters {
    is_controller: boolean = true;      // Controller or DRAM role
    voltage: voltage = 1.2V;            // JEDEC standard voltage
  }
  
  pins {
    // Address signals
    A[0:13]: is_controller ? out(voltage) : in(voltage);         // Row address bits A0-A13
    BA[0:1]: is_controller ? out(voltage) : in(voltage);         // Bank address
    BG[0:1]: is_controller ? out(voltage) : in(voltage);         // Bank group
    PAR: is_controller ? out(voltage) : in(voltage);             // Command/address parity
  }
  
  constraints {
    setup(A to posedge CK): >100ps;
    hold(A to posedge CK): >50ps;
  }
}

// Define DDR4 Clock Interface
interface DDR4Clock {
  parameters {
    is_controller: boolean = true;      // Controller or DRAM role
    voltage: voltage = 1.2V;            // JEDEC standard voltage
    data_rate: frequency = 3200MT/s;    // Transfer rate
  }
  
  pins {
    // Clock signals
    CK: is_controller ? out(voltage) : in(voltage);             // Primary clock
    CK_n: is_controller ? out(voltage) : in(voltage);           // Complementary clock
  }
  
  constraints {
    clock_period: 1/(data_rate/2);  // Clock period (e.g., 625ps for DDR4-3200)
    duty_cycle: 45-55%;
  }
}

// Component definition for DDR4 Memory Controller
component DDR4Controller {
  parameters {
    width: int = 64;                     // Data width (64-bit typical)
    ranks: int = 2;                      // Number of ranks
    data_rate: frequency = 3200MT/s;     // Transfer speed
  }
  
  // Interfaces
  interfaces {
    DATA: interface DDR4Data {
      is_controller: true;
      width: width;
    }
    
    CMD: interface DDR4Command {
      is_controller: true;
      ranks: ranks;
    }
    
    ADDR: interface DDR4Address {
      is_controller: true;
    }
    
    CLK: interface DDR4Clock {
      is_controller: true;
      data_rate: data_rate;
    }
  }
  
  // Power pins defined separately from interfaces
  pins {
    VDD: power(1.2V);                   // Core voltage
    VDDQ: power(1.2V);                  // I/O voltage
    VSS: ground;                         // Ground
  }
}

// Component definition for DDR4 DRAM
component DDR4DRAM {
  parameters {
    density: enum {4Gb, 8Gb, 16Gb} = 8Gb;  // Density options
    width: int = 16;                       // Data bus width (x4, x8, or x16)
    speed_bin: enum {2400, 2666, 3200} = 3200;  // Speed bin
  }
  
  // Interfaces
  interfaces {
    DATA: interface DDR4Data {
      is_controller: false;
      width: width;
    }
    
    CMD: interface DDR4Command {
      is_controller: false;
      ranks: 1;
    }
    
    ADDR: interface DDR4Address {
      is_controller: false;
    }
    
    CLK: interface DDR4Clock {
      is_controller: false;
      data_rate: speed_bin;
    }
  }
  
  // Power pins defined separately from interfaces
  pins {
    VDD: power(1.2V);     // Core voltage
    VDDQ: power(1.2V);    // I/O voltage
    VSS: ground;          // Ground
    VREF: power(0.6V);    // Reference voltage
  }
}

// Board definition with DDR4 subsystem
board DDR4System {
  components {
    // DDR4 controller
    component DDR4Controller CTRL {
      width: 64;
      ranks: 2;
    }
    
    // Four x16 DRAMs to create a 64-bit data bus
    component DDR4DRAM DRAM[0:3] {
      width: 16;
    }
    
    // Termination resistors
    component Resistor RTERM[0:20] {
      value: 40Ω;
    }
  }
  
  // Net definitions
  nets {
    VDD_1V2: power(1.2V);
    VREF_DDR4: power(0.6V);
    GND: ground;
  }
  
  connections {
    // Power connections handled separately from interfaces
    CTRL.VDD -> VDD_1V2;
    CTRL.VDDQ -> VDD_1V2;
    CTRL.VSS -> GND;
    
    DRAM[0:3].VDD -> VDD_1V2;
    DRAM[0:3].VDDQ -> VDD_1V2;
    DRAM[0:3].VSS -> GND;
    DRAM[0:3].VREF -> VREF_DDR4;
    
    // Data interface connections with swizzle maps
    // Define swizzle maps for data bits
    swizzle_map DRAM0_MAP {
      CTRL.DATA.DQ[0:7] -> DRAM[0].DATA.DQ[2,3,1,0,6,7,5,4];
      CTRL.DATA.DQ[8:15] -> DRAM[0].DATA.DQ[10,11,9,8,14,15,13,12];
    }
    
    swizzle_map DRAM1_MAP {
      CTRL.DATA.DQ[16:23] -> DRAM[1].DATA.DQ[2,3,1,0,6,7,5,4];
      CTRL.DATA.DQ[24:31] -> DRAM[1].DATA.DQ[10,11,9,8,14,15,13,12];
    }
    
    // Connect data interfaces with swizzle maps
    CTRL.DATA <=> DRAM[0].DATA {
      DQ: apply DRAM0_MAP;
      DQS[0] -> DQS[0];
      DQS[1] -> DQS[1];
      DQS_n[0] -> DQS_n[0];
      DQS_n[1] -> DQS_n[1];
    };
    
    CTRL.DATA <=> DRAM[1].DATA {
      DQ: apply DRAM1_MAP;
      DQS[2] -> DQS[0];
      DQS[3] -> DQS[1];
      DQS_n[2] -> DQS_n[0];
      DQS_n[3] -> DQS_n[1];
    };
    
    // Similar connections for DRAM2 and DRAM3 (omitted for brevity)
    
    // Connect address interface to all DRAMs
    CTRL.ADDR <=> {DRAM[0:3].ADDR};
    
    // Connect clock interface to all DRAMs
    CTRL.CLK <=> {DRAM[0:3].CLK};
    
    // Connect command interface with rank-specific signals
    CTRL.CMD <=> {DRAM[0:3].CMD} {
      // Override chip select lines per rank
      CS_n[0]: [DRAM[0].CMD.CS_n[0], DRAM[1].CMD.CS_n[0]];  // Rank 0
      CS_n[1]: [DRAM[2].CMD.CS_n[0], DRAM[3].CMD.CS_n[0]];  // Rank 1
      
      // Broadcast all other signals
      ACT_n: broadcast;
      RAS_n_A16: broadcast;
      CAS_n_A15: broadcast;
      WE_n_A14: broadcast;
    };
    
    // Add termination resistors to address/command signals
    CTRL.ADDR.A[0:13] -[RTERM[0:13]]-> GND;
  }
}
