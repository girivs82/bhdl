// Valid Physical Literal for use in parameters
1kOhm;

board TestBoard {
  parameters {
    N: integer = 4;
  }

  components {
    // --- Instantiations ---

    // Parameter-less (Requires {})
    MCU P1 {};
    Device DEVICE1 {};

    // With parameters
    Capacitor C1 { value = 10uF, voltage = 16Vdc };

    // Inside generate block (parameter-less)
    generate for idx in (0..1) {
       LED L[idx] {};
    }

    // Inside generate block (with parameters)
    generate for i in (0..N-1) {
       Resistor R_GEN[i] { Value = 1kOhm };
    }
  }

  connections {
    // Basic connection to check endpoints are still okay
    P1.out -> DEVICE1.in;
    C1.p1 -> GND;
  }
} end board;

// Component Instantiation with Generate
// -------------------------------------
component Resistor end component;

board GeneratedCircuit {
  parameters {
    NUM_RESISTORS = 4;
  }
  components {
    // Generate component instantiations
    generate i in (0 .. NUM_RESISTORS-1) {
      Resistor R_GEN[i](Value=1k); // Instantiation with () parameters
      Series R_SER[i]{Value=1k};   // Instantiation with {} parameters
    }
  }
  pins {
      pin Vin, Vout;
  }
  ports {
      port P1 : in signal;
      port P2 : out signal;
  }
  connections {
      // Connect generated components
      generate i in (0 .. NUM_RESISTORS-2) {
          R_GEN[i].p2 -> R_GEN[i+1].p1;
      }
      Vin -> R_GEN[0].p1;
      R_GEN[NUM_RESISTORS-1].p2 -> Vout;
      P1 -> R_SER[0].p_in;
      R_SER[NUM_RESISTORS-1].p_out -> P2;
  }
} end board;
